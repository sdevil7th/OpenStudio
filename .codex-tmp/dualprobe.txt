            handoffRack.getLastLoadError());
        return juce::var(value);
    };

    auto stressStatus = [] (const juce::var& stressResult)

                return juce::String("not_asserted");

        }

    auto runDualMonoNAMGraphContractProbe = [&] ()
    {
        auto* value = new juce::DynamicObject();

            "modelPath", a1ModelFile.getFullPathName());

            a1ModelFile.existsAsFile();
        value->setProperty("available", available);
        if (! available)
        {
            value->setProperty("pass", false);

        }

        constexpr double testSampleRate = 48000.0;
        constexpr int testBlockSize = 8;
        constexpr int totalSamples = 4096;
        const auto configureModelRack = [&] (
            S13NAMRack& rack,
            bool loadPedal)
        {
            rack.prepareToPlay(
                testSampleRate, testBlockSize);
            configureNeutralRack(rack);
            rack.inputMode.store(
                2.0f, std::memory_order_relaxed);
            rack.pedalCalibrationMode.store(
                0.0f, std::memory_order_relaxed);
            rack.ampCalibrationMode.store(
                0.0f, std::memory_order_relaxed);
            rack.pedalMix.store(
                loadPedal ? 1.0f : 0.0f,
                std::memory_order_relaxed);
            rack.ampEnabled.store(
                1.0f, std::memory_order_relaxed);
            rack.ampMix.store(
                1.0f, std::memory_order_relaxed);
            const bool pedalLoaded =
                ! loadPedal
                || rack.loadPedalModel(
                    a1ModelFile.getFullPathName());
            const bool ampLoaded =
                rack.loadAmpModel(
                    a1ModelFile.getFullPathName());
            // A reset publishes the already-prepared Amp owner directly for
            // this offline probe instead of spending its first blocks in the
            // live model-handoff envelope.
            rack.reset();
            return pedalLoaded && ampLoaded;
        };

        S13NAMRack ownershipRack;
        const bool ownershipLoaded =
            configureModelRack(ownershipRack, true);
        const auto* const pedalPrimary =
            ownershipRack.pedalModel != nullptr
                ? ownershipRack.pedalModel->dsp.get()
                : nullptr;
        const auto* const pedalSecondary =
            ownershipRack.pedalModel != nullptr
                    && ownershipRack.pedalModel
                           ->dualMonoLane != nullptr
                ? ownershipRack.pedalModel
                      ->dualMonoLane->dsp.get()
                : nullptr;
        const auto* const ampPrimary =
            ownershipRack.ampModel != nullptr
                ? ownershipRack.ampModel->dsp.get()
                : nullptr;
        const auto* const ampSecondary =
            ownershipRack.ampModel != nullptr
                    && ownershipRack.ampModel
                           ->dualMonoLane != nullptr
                ? ownershipRack.ampModel
                      ->dualMonoLane->dsp.get()
                : nullptr;
        const bool fourDistinctGraphs =
            ownershipLoaded
            && pedalPrimary != nullptr
            && pedalSecondary != nullptr
            && ampPrimary != nullptr
            && ampSecondary != nullptr
            && pedalPrimary != pedalSecondary
            && pedalPrimary != ampPrimary
            && pedalPrimary != ampSecondary
            && pedalSecondary != ampPrimary
            && pedalSecondary != ampSecondary
            && ampPrimary != ampSecondary;

        const auto render = [&] (
            S13NAMRack& rack,
            bool antiPhase,
            bool identicalStereo,
            const std::array<int, 4>& partitions,
            float sourceGain)
        {
            juce::AudioBuffer<float> rendered(
                2, totalSamples);
            for (int sample = 0;
                 sample < totalSamples;
                 ++sample)
            {
                const float time =
                    static_cast<float>(sample)
                    / static_cast<float>(
                        testSampleRate);
                const float input =
                    sourceGain
                    * (0.12f
                    * std::sin(
                        juce::MathConstants<float>::twoPi
                        * 173.0f * time)
                    + 0.035f
                    * std::sin(
                        juce::MathConstants<float>::twoPi
                        * 619.0f * time));
                rendered.setSample(0, sample, input);
                rendered.setSample(
                    1,
                    sample,
                    identicalStereo
                        ? input
                        : (antiPhase ? -input : 0.0f));
            }

            juce::MidiBuffer midi;
            int cursor = 0;
            int partitionIndex = 0;
            while (cursor < totalSamples)
            {
                const int requested =
                    partitions[static_cast<size_t>(
                        partitionIndex
                        % static_cast<int>(
                            partitions.size()))];
                const int blockSamples =
                    juce::jmin(
                        totalSamples - cursor,
                        juce::jmax(1, requested));
                std::array<float*, 2> channelPointers {
                    rendered.getWritePointer(0, cursor),
                    rendered.getWritePointer(1, cursor)
                };
                juce::AudioBuffer<float> block(
                    channelPointers.data(),
                    2,
                    blockSamples);
                rack.processBlock(block, midi);
                cursor += blockSamples;
                ++partitionIndex;
            }
            return rendered;
        };

        const auto measureStereo = [] (
            const juce::AudioBuffer<float>& rendered)
        {
            std::array<double, 3> measurement {};
            for (int sample = 0;
                 sample < rendered.getNumSamples();
                 ++sample)
            {
                const float left =
                    rendered.getSample(0, sample);
                const float right =
                    rendered.getSample(1, sample);
                if (! std::isfinite(left)
                    || ! std::isfinite(right))
                {
                    measurement[2] += 1.0;
                    continue;
                }
                const double side =
                    0.5
                    * static_cast<double>(
                        left - right);
                measurement[0] += side * side;
                measurement[1] = juce::jmax(
                    measurement[1],
                    static_cast<double>(
                        std::abs(left - right)));
            }
            measurement[0] = std::sqrt(
                measurement[0]
                / static_cast<double>(
                    juce::jmax(
                        1,
                        rendered.getNumSamples())));
            return measurement;
        };

        S13NAMRack dualAntiRack;
        S13NAMRack singleAntiRack;
        const bool antiRacksLoaded =
            configureModelRack(dualAntiRack, false)
            && configureModelRack(singleAntiRack, false);
        singleAntiRack.inputMode.store(
            0.0f, std::memory_order_relaxed);
        singleAntiRack.reset();
        const auto dualAnti = render(
            dualAntiRack,
            true,
            false,
            { 8, 8, 8, 8 },
            1.0f);
        const auto singleAnti = render(
            singleAntiRack,
            true,
            false,
            { 8, 8, 8, 8 },
            1.0f);
        const auto dualAntiMeasurement =
            measureStereo(dualAnti);
        const auto singleAntiMeasurement =
            measureStereo(singleAnti);
        const bool antiPhaseRoutingPass =
            antiRacksLoaded
            && dualAntiMeasurement[2] == 0.0
            && singleAntiMeasurement[2] == 0.0
            && dualAntiMeasurement[0] > 1.0e-4
            && singleAntiMeasurement[0] < 1.0e-8;
        const bool parallelLatencyPass =
            antiRacksLoaded
            && dualAntiRack.getLatencySamples()
                == singleAntiRack.getLatencySamples();

        dualAntiRack.reset();
        const auto leftOnlyOutput = render(
            dualAntiRack,
            false,
            false,
            { 8, 8, 8, 8 },
            1.0f);
        dualAntiRack.reset();
        const auto silenceOutput = render(
            dualAntiRack,
            false,
            false,
            { 8, 8, 8, 8 },
            0.0f);
        float rightLaneIsolationDifference = 0.0f;
        float leftOnlyPeak = 0.0f;
        for (int sample = 0;
             sample < totalSamples;
             ++sample)
        {
            rightLaneIsolationDifference =
                juce::jmax(
                    rightLaneIsolationDifference,
                    std::abs(
                        leftOnlyOutput.getSample(1, sample)
                        - silenceOutput.getSample(1, sample)));
            leftOnlyPeak = juce::jmax(
                leftOnlyPeak,
                std::abs(
                    leftOnlyOutput.getSample(0, sample)));
        }
        const bool laneIsolationPass =
            leftOnlyPeak > 1.0e-4f
            && rightLaneIsolationDifference <= 1.0e-7f;

        S13NAMRack identicalRack;
        const bool identicalLoaded =
            configureModelRack(identicalRack, false);
        const auto identicalOutput = render(
            identicalRack,
            false,
            true,
            { 8, 8, 8, 8 },
            1.0f);
        const auto identicalMeasurement =
            measureStereo(identicalOutput);
        const bool identicalStereoParity =
            identicalLoaded
            && identicalMeasurement[2] == 0.0
            && identicalMeasurement[1] <= 1.0e-6;

        S13NAMRack fixedRack;
        S13NAMRack unevenRack;
        const bool partitionRacksLoaded =
            configureModelRack(fixedRack, false)
            && configureModelRack(unevenRack, false);
        const auto fixedOutput = render(
            fixedRack,
            false,
            true,
            { 8, 8, 8, 8 },
            1.0f);
        const auto unevenOutput = render(
            unevenRack,
            false,
            true,
            { 3, 5, 7, 1 },
            1.0f);
        float partitionMaximumDifference = 0.0f;
        int partitionNonFiniteCount = 0;
        for (int channel = 0;
             channel < 2;
             ++channel)
        {
            for (int sample = 0;
                 sample < totalSamples;
                 ++sample)
            {
                const float fixed =
                    fixedOutput.getSample(
                        channel, sample);
                const float uneven =
                    unevenOutput.getSample(
                        channel, sample);
                if (! std::isfinite(fixed)
                    || ! std::isfinite(uneven))
                {
                    ++partitionNonFiniteCount;
                    continue;
                }
                partitionMaximumDifference =
                    juce::jmax(
                        partitionMaximumDifference,
                        std::abs(fixed - uneven));
            }
        }
        const bool partitionPass =
            partitionRacksLoaded
            && partitionNonFiniteCount == 0
            && partitionMaximumDifference
                <= 1.0e-5f;

        // Reproduce a fault raised by the right graph from inside its own
        // process() call. The pair-level pre-check has already passed at that
        // point, so this specifically guards against committing the completed
        // left wet lane beside the right lane's fault fallback.
        class RuntimeFaultProbeDSP final : public nam::DSP
        {
        public:
            RuntimeFaultProbeDSP(
                std::atomic<bool>& faultFlagToUse,
                double expectedSampleRate)
                : nam::DSP(1, 1, expectedSampleRate),
                  faultFlag(faultFlagToUse)
            {
            }

            void process(
                float** input,
                float** output,
                const int numFrames) override
            {
                juce::ignoreUnused(input);
                if (output != nullptr && output[0] != nullptr)
                {
                    for (int sample = 0;
                         sample < numFrames;
                         ++sample)
                    {
                        output[0][sample] =
                            std::numeric_limits<float>::quiet_NaN();
                    }
                }
                faultFlag.store(
                    true, std::memory_order_relaxed);
            }

        private:
            std::atomic<bool>& faultFlag;
        };

        S13NAMRack runtimeFaultRack;
        S13NAMRack runtimeFaultDryReferenceRack;
        S13NAMRack runtimeFaultWetControlRack;
        const bool runtimeFaultRacksLoaded =
            configureModelRack(runtimeFaultRack, false)
            && configureModelRack(
                runtimeFaultDryReferenceRack, false)
            && configureModelRack(
                runtimeFaultWetControlRack, false);
        const bool runtimeFaultGraphsReady =
            runtimeFaultRacksLoaded
            && runtimeFaultRack.ampModel != nullptr
            && runtimeFaultRack.ampModel
                   ->dualMonoLane != nullptr
            && runtimeFaultDryReferenceRack.ampModel
                != nullptr
            && runtimeFaultDryReferenceRack.ampModel
                   ->dualMonoLane != nullptr
            && runtimeFaultWetControlRack.ampModel
                != nullptr
            && runtimeFaultWetControlRack.ampModel
                   ->dualMonoLane != nullptr;
        juce::AudioBuffer<float> runtimeFaultBlock(
            2, testBlockSize);
        juce::AudioBuffer<float> runtimeFaultDryReferenceBlock(
            2, testBlockSize);
        juce::AudioBuffer<float> runtimeFaultWetControlBlock(
            2, testBlockSize);
        runtimeFaultBlock.clear();
        runtimeFaultDryReferenceBlock.clear();
        runtimeFaultWetControlBlock.clear();
        const auto fillRuntimeFaultBlock = [&] (
            juce::AudioBuffer<float>& block,
            int blockIndex)
        {
            for (int sample = 0;
                 sample < testBlockSize;
                 ++sample)
            {
                const int absoluteSample =
                    blockIndex * testBlockSize + sample;
                const float time =
                    static_cast<float>(absoluteSample)
                    / static_cast<float>(testSampleRate);
                const float input =
                    0.09f
                    * (std::sin(
                           juce::MathConstants<float>::twoPi
                           * 197.0f * time)
                       + 0.31f
                           * std::sin(
                               juce::MathConstants<float>::twoPi
                               * 733.0f * time));
                block.setSample(0, sample, input);
                block.setSample(1, sample, input);
            }
        };
        juce::MidiBuffer runtimeFaultMidi;
        constexpr int runtimeFaultWarmupBlocks = 96;
        if (runtimeFaultGraphsReady)
        {
            for (int blockIndex = 0;
                 blockIndex < runtimeFaultWarmupBlocks;
                 ++blockIndex)
            {
                fillRuntimeFaultBlock(
                    runtimeFaultBlock, blockIndex);
                fillRuntimeFaultBlock(
                    runtimeFaultDryReferenceBlock,
                    blockIndex);
                fillRuntimeFaultBlock(
                    runtimeFaultWetControlBlock,
                    blockIndex);
                runtimeFaultRack.processBlock(
                    runtimeFaultBlock, runtimeFaultMidi);
                runtimeFaultDryReferenceRack.processBlock(
                    runtimeFaultDryReferenceBlock,
                    runtimeFaultMidi);
                runtimeFaultWetControlRack.processBlock(
                    runtimeFaultWetControlBlock,
                    runtimeFaultMidi);
            }

            auto& injectedRightLane =
                *runtimeFaultRack.ampModel->dualMonoLane;
            injectedRightLane.dsp =
                std::make_unique<RuntimeFaultProbeDSP>(
                    injectedRightLane.processFaulted,
                    testSampleRate);
            runtimeFaultDryReferenceRack.ampModel
                ->processFaulted.store(
                    true, std::memory_order_relaxed);
            runtimeFaultDryReferenceRack.ampModel
                ->dualMonoLane->processFaulted.store(
                    true, std::memory_order_relaxed);

            fillRuntimeFaultBlock(
                runtimeFaultBlock,
                runtimeFaultWarmupBlocks);
            fillRuntimeFaultBlock(
                runtimeFaultDryReferenceBlock,
                runtimeFaultWarmupBlocks);
            fillRuntimeFaultBlock(
                runtimeFaultWetControlBlock,
                runtimeFaultWarmupBlocks);
            runtimeFaultRack.processBlock(
                runtimeFaultBlock, runtimeFaultMidi);
            runtimeFaultDryReferenceRack.processBlock(
                runtimeFaultDryReferenceBlock,
                runtimeFaultMidi);
            runtimeFaultWetControlRack.processBlock(
                runtimeFaultWetControlBlock,
                runtimeFaultMidi);
        }

        float runtimeFaultDryMaximumDifference = 0.0f;
        float runtimeFaultWetControlDifference = 0.0f;
        float runtimeFaultInterchannelDifference = 0.0f;
        int runtimeFaultNonFiniteCount = 0;
        for (int channel = 0; channel < 2; ++channel)
        {
            for (int sample = 0;
                 sample < testBlockSize;
                 ++sample)
            {
                const float faultOutput =
                    runtimeFaultBlock.getSample(
                        channel, sample);
                const float dryReference =
                    runtimeFaultDryReferenceBlock.getSample(
                        channel, sample);
                const float wetControl =
                    runtimeFaultWetControlBlock.getSample(
                        channel, sample);
                if (! std::isfinite(faultOutput)
                    || ! std::isfinite(dryReference)
                    || ! std::isfinite(wetControl))
                {
                    ++runtimeFaultNonFiniteCount;
                    continue;
                }
                runtimeFaultDryMaximumDifference =
                    juce::jmax(
                        runtimeFaultDryMaximumDifference,
                        std::abs(
                            faultOutput - dryReference));
                runtimeFaultWetControlDifference =
                    juce::jmax(
                        runtimeFaultWetControlDifference,
                        std::abs(
                            wetControl - dryReference));
                runtimeFaultInterchannelDifference =
                    juce::jmax(
                        runtimeFaultInterchannelDifference,
                        std::abs(
                            runtimeFaultBlock.getSample(
                                0, sample)
                            - runtimeFaultBlock.getSample(
                                1, sample)));
            }
        }
        const bool runtimeFaultRaised =
            runtimeFaultGraphsReady
            && runtimeFaultRack.ampModel
                   ->processFaulted.load(
                       std::memory_order_relaxed)
            && runtimeFaultRack.ampModel
                   ->dualMonoLane->processFaulted.load(
                       std::memory_order_relaxed);
        const bool runtimeFaultAtomicFallbackPass =
            runtimeFaultGraphsReady
            && runtimeFaultRaised
            && runtimeFaultNonFiniteCount == 0
            && runtimeFaultDryMaximumDifference
                <= 1.0e-6f
            && runtimeFaultInterchannelDifference
                <= 1.0e-6f
            && runtimeFaultWetControlDifference
                > 1.0e-5f;

        const auto renderModeChange = [&] (
            S13NAMRack& rack,
            const std::array<int, 4>& partitions)
        {
            rack.inputMode.store(
                0.0f, std::memory_order_relaxed);
            rack.reset();
            juce::AudioBuffer<float> rendered(
                2, totalSamples);
            for (int sample = 0;
                 sample < totalSamples;
                 ++sample)
            {
                const float input =
                    0.11f
                    * std::sin(
                        juce::MathConstants<float>::twoPi
                        * 211.0f
                        * static_cast<float>(sample)
                        / static_cast<float>(
                            testSampleRate));
                rendered.setSample(0, sample, input);
                rendered.setSample(1, sample, input);
            }

            constexpr int modeChangeSample = 512;
            juce::MidiBuffer midi;
            int cursor = 0;
            int partitionIndex = 0;
            while (cursor < totalSamples)
            {
                if (cursor == modeChangeSample)
                {
                    rack.inputMode.store(
                        2.0f,
                        std::memory_order_relaxed);
                }
                const int requested =
                    partitions[static_cast<size_t>(
                        partitionIndex
                        % static_cast<int>(
                            partitions.size()))];
                const int blockSamples =
                    juce::jmin(
                        totalSamples - cursor,
                        juce::jmax(1, requested));
                std::array<float*, 2> pointers {
                    rendered.getWritePointer(0, cursor),
                    rendered.getWritePointer(1, cursor)
                };
                juce::AudioBuffer<float> block(
                    pointers.data(),
                    2,
                    blockSamples);
                rack.processBlock(block, midi);
                cursor += blockSamples;
                ++partitionIndex;
            }
            return rendered;
        };

        const auto fixedModeChange =
            renderModeChange(
                fixedRack,
                { 8, 8, 8, 8 });
        const auto unevenModeChange =
            renderModeChange(
                unevenRack,
                { 3, 5, 7, 1 });
        float modeChangePartitionDifference = 0.0f;
        float modeChangePeak = 0.0f;
        float modeChangeMaximumStep = 0.0f;
        float modeChangeSteadyReferenceMaximumStep = 0.0f;
        float modeChangeTransitionMaximumStep = 0.0f;
        int modeChangeNonFiniteCount = 0;
        int currentMutedRun = 0;
        int longestMutedRun = 0;
        std::array<float, 2> previous {};
        for (int sample = 0;
             sample < totalSamples;
             ++sample)
        {
            bool frameMuted = true;
            for (int channel = 0;
                 channel < 2;
                 ++channel)
            {
                const float fixed =
                    fixedModeChange.getSample(
                        channel, sample);
                const float uneven =
                    unevenModeChange.getSample(
                        channel, sample);
                if (! std::isfinite(fixed)
                    || ! std::isfinite(uneven))
                {
                    ++modeChangeNonFiniteCount;
                    continue;
                }
                modeChangePartitionDifference =
                    juce::jmax(
                        modeChangePartitionDifference,
                        std::abs(fixed - uneven));
                modeChangePeak = juce::jmax(
                    modeChangePeak,
                    std::abs(fixed));
                if (sample > 0)
                {
                    const float step = std::abs(
                        fixed
                        - previous[static_cast<size_t>(channel)]);
                    modeChangeMaximumStep = juce::jmax(
                        modeChangeMaximumStep, step);
                    // Use the already-settled, pre-change NAM waveform as the
                    // natural high-gain step reference.  Measuring the raw
                    // output against an arbitrary percentage of peak wrongly
                    // classifies ordinary distorted-guitar edges as routing
                    // clicks.  The transition gate below measures only excess
                    // step energy above this same-render reference.
                    if (sample >= 128 && sample < 512)
                    {
                        modeChangeSteadyReferenceMaximumStep =
                            juce::jmax(
                                modeChangeSteadyReferenceMaximumStep,
                                step);
                    }
                    else if (sample >= 512)
                    {
                        modeChangeTransitionMaximumStep =
                            juce::jmax(
                                modeChangeTransitionMaximumStep,
                                step);
                    }
                }
                previous[static_cast<size_t>(channel)] =
                    fixed;
                frameMuted = frameMuted
                    && std::abs(fixed) <= 1.0e-8f;
            }
            currentMutedRun = frameMuted
                ? currentMutedRun + 1
                : 0;
            longestMutedRun = juce::jmax(
                longestMutedRun,
                currentMutedRun);
        }
        const float modeChangeMaximumStepRatio =
            modeChangeMaximumStep
            / juce::jmax(0.05f, modeChangePeak);
        const float modeChangeExcessStep = juce::jmax(
            0.0f,
            modeChangeTransitionMaximumStep
                - modeChangeSteadyReferenceMaximumStep);
        const float modeChangeExcessStepRatio =
            modeChangeExcessStep
            / juce::jmax(0.05f, modeChangePeak);
        const bool modeChangeTransitionPass =
            modeChangeNonFiniteCount == 0
            && modeChangePartitionDifference
                <= 1.0e-5f
            && modeChangeSteadyReferenceMaximumStep > 1.0e-6f
            && modeChangeExcessStepRatio <= 0.10f
            && longestMutedRun
                >= juce::roundToInt(
                    testSampleRate * 0.023);

        juce::MemoryBlock savedState;
        ownershipRack.getStateInformation(savedState);
        S13NAMRack restoredRack;
        restoredRack.prepareToPlay(
            testSampleRate, testBlockSize);
        const bool restored =
            restoredRack.restoreProjectStateInformation(
                savedState.getData(),
                static_cast<int>(
                    savedState.getSize()));
        const bool stateRoundTrip =
            restored
            && std::abs(
                   restoredRack.inputMode.load(
                       std::memory_order_relaxed)
                   - 2.0f)
                <= 1.0e-6f
            && restoredRack.pedalModel != nullptr
            && restoredRack.pedalModel
                   ->dualMonoLane != nullptr
            && restoredRack.ampModel != nullptr
            && restoredRack.ampModel
                   ->dualMonoLane != nullptr;

        auto* const originalAmpOwner =
            ownershipRack.ampModel.get();
        auto* const originalAmpLane =
            ownershipRack.ampModel != nullptr
                ? ownershipRack.ampModel
                      ->dualMonoLane.get()
                : nullptr;
        const bool invalidReplacementRejected =
            ! ownershipRack.loadAmpModel(
                invalidNamFile.getFullPathName());
        const bool failedLoadPreservesPair =
            invalidReplacementRejected
            && ownershipRack.ampModel.get()
                == originalAmpOwner
            && ownershipRack.ampModel != nullptr
            && ownershipRack.ampModel
                   ->dualMonoLane.get()
                == originalAmpLane;

        S13NAMRack singleLaneFallbackRack;
        const bool singleLaneFallbackLoaded =
            configureModelRack(
                singleLaneFallbackRack, false);
        if (singleLaneFallbackRack.ampModel != nullptr)
        {
            singleLaneFallbackRack.ampModel
                ->dualMonoPreparationWarning =
                    "synthetic secondary-lane preparation failure";
            singleLaneFallbackRack.ampModel
                ->dualMonoLane.reset();
        }
        singleLaneFallbackRack.reset();
        const auto singleLaneFallbackOutput = render(
            singleLaneFallbackRack,
            false,
            true,
            { 8, 8, 8, 8 },
            1.0f);
        bool singleLaneFallbackFinite = true;
        float singleLaneFallbackPeak = 0.0f;
        for (int channel = 0;
             channel < singleLaneFallbackOutput.getNumChannels();
             ++channel)
        {
            for (int sample = 0;
                 sample < singleLaneFallbackOutput.getNumSamples();
                 ++sample)
            {
                const float sampleValue =
                    singleLaneFallbackOutput.getSample(
                        channel, sample);
                singleLaneFallbackFinite =
                    singleLaneFallbackFinite
                    && std::isfinite(sampleValue);
                singleLaneFallbackPeak = juce::jmax(
                    singleLaneFallbackPeak,
                    std::abs(sampleValue));
            }
        }
        const auto singleLaneFallbackDiagnostics =
            singleLaneFallbackRack.getDiagnosticState();
        const bool singleLaneFallbackPass =
            singleLaneFallbackLoaded
            && singleLaneFallbackRack.hasAmpModel()
            && singleLaneFallbackFinite
            && singleLaneFallbackPeak > 1.0e-5f
            && ! static_cast<bool>(
